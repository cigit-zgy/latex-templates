% Page geometry plus project-wide heading and running-head tokens.
\geometry{
  letterpaper,
  inner=31mm,
  outer=31mm,
  top=25mm,
  bottom=30mm,
  headheight=14pt,
  headsep=8mm,
  footskip=12mm
}

\setlength{\parindent}{1.2em}
\setlength{\parskip}{0pt}
\linespread{1.08}
\raggedbottom
\setcounter{tocdepth}{2}
\setcounter{secnumdepth}{3}

% Thesis-only chapter opening. Number and title share the same colour.
\titleformat{\chapter}[display]
  {\normalfont\AcademicHeadingFont\bfseries\color{AcademicChapter}}
  {\makebox[\textwidth][r]{\sffamily\bfseries\fontsize{62}{62}\selectfont\thechapter}}
  {-8pt}
  {\fontsize{22}{28}\selectfont\HeadingText}
  [\vspace{0.7ex}{\color{AcademicRule}\titlerule[0.5pt]}]
\titleformat{name=\chapter,numberless}[display]
  {\normalfont\AcademicHeadingFont\bfseries\color{AcademicChapter}}
  {}
  {0pt}
  {\fontsize{22}{28}\selectfont\HeadingText}
  [\vspace{0.7ex}{\color{AcademicRule}\titlerule[0.5pt]}]
\titlespacing*{\chapter}{0pt}{-6pt}{32pt}

% Shared three-level hierarchy. Number and title inherit exactly the same style.
\titleformat{\section}
  {\normalfont\AcademicHeadingFont\bfseries\fontsize{16}{20}\selectfont\color{AcademicLevelOne}}
  {\thesection}{0.75em}{}
\titlespacing*{\section}{0pt}{3.0ex plus .8ex minus .2ex}{1.1ex}

\titleformat{\subsection}
  {\normalfont\AcademicHeadingFont\bfseries\fontsize{13}{16}\selectfont\color{AcademicLevelTwo}}
  {\thesubsection}{0.70em}{}
\titlespacing*{\subsection}{0pt}{2.4ex plus .6ex minus .2ex}{0.8ex}

\titleformat{\subsubsection}
  {\normalfont\AcademicHeadingFont\bfseries\fontsize{11}{14}\selectfont\color{AcademicLevelThree}}
  {\thesubsubsection}{0.65em}{}
\titlespacing*{\subsubsection}{0pt}{2.0ex plus .5ex minus .2ex}{0.6ex}

% Shared running-head treatment: muted, sans-serif, no rule.
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\fancyhead[LE]{\small\color{AcademicMuted}\thepage}
\fancyhead[RE]{\small\AcademicHeaderFont\color{AcademicMuted}\nouppercase{\rightmark}}
\fancyhead[LO]{\small\AcademicHeaderFont\color{AcademicMuted}\nouppercase{\leftmark}}
\fancyhead[RO]{\small\color{AcademicMuted}\thepage}
\renewcommand{\chaptermark}[1]{\markboth{#1}{}}
\renewcommand{\sectionmark}[1]{\markright{#1}}

\fancypagestyle{plain}{
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \fancyfoot[C]{\small\color{AcademicMuted}\thepage}
}
