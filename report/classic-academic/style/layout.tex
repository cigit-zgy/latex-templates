% Single-sided report geometry, heading hierarchy, and running heads.

\geometry{
  letterpaper,
  left=31mm,
  right=31mm,
  top=25mm,
  bottom=28mm,
  headheight=14pt,
  headsep=8mm,
  footskip=12mm
}

\setlength{\parindent}{1.2em}
\setlength{\parskip}{0pt}
\linespread{1.08}
\raggedbottom

\setcounter{tocdepth}{2}
\setcounter{secnumdepth}{3}

% Reports keep numbered structure internally and in the TOC, but do not print
% the large display numeral at the upper right of each section heading.
\titleformat{\section}[display]
  {\normalfont}
  {}
  {0pt}
  {\color{AcademicText}\AcademicHeadingFont\fontsize{18}{23}\selectfont\HeadingText}
  [\vspace{0.5ex}{\color{AcademicRule}\titlerule}]
\titlespacing*{\section}{0pt}{3.8ex plus 1ex minus .2ex}{2.4ex}

\titleformat{\subsection}
  {\normalfont\color{AcademicAccent}\AcademicHeadingFont\large}
  {\thesubsection}
  {0.8em}
  {}
\titlespacing*{\subsection}{0pt}{2.8ex plus .8ex minus .2ex}{0.9ex}

\titleformat{\subsubsection}
  {\normalfont\bfseries}
  {\thesubsubsection}
  {0.7em}
  {}

\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.45pt}
\renewcommand{\footrulewidth}{0pt}
\fancyhead[L]{\small\AcademicHeaderFont\ReportShortTitle}
\fancyhead[R]{\small\thepage}

\fancypagestyle{plain}{
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \fancyfoot[C]{\thepage}
}
