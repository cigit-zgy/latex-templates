% Single-sided report geometry plus project-wide heading and running-head tokens.
% Body-text geometry follows conservative book-typography guidance: a moderate
% measure and leading near 142% of the nominal 11 pt body size.
\geometry{
  letterpaper,
  left=32mm,
  right=32mm,
  top=26mm,
  bottom=29mm,
  headheight=14pt,
  headsep=8mm,
  footskip=12mm
}

\setlength{\parindent}{1.2em}
\setlength{\parskip}{0pt}
% 11 pt article has a 13.6 pt nominal baseline; 1.15 gives about 15.6 pt,
% i.e. roughly 142% of body point size, near the upper end of the common
% 120--145% book-reading range and intentionally less dense for technical prose.
\linespread{1.15}
\raggedbottom
\setcounter{tocdepth}{2}
\setcounter{secnumdepth}{3}

% Shared three-level hierarchy. Number and title inherit exactly the same style.
\titleformat{\section}
  {\normalfont\AcademicHeadingFont\bfseries\fontsize{16}{20}\selectfont\color{AcademicLevelOne}}
  {\thesection}{0.75em}{}
\titlespacing*{\section}{0pt}{3.2ex plus .8ex minus .2ex}{1.25ex}

\titleformat{\subsection}
  {\normalfont\AcademicHeadingFont\bfseries\fontsize{13}{16}\selectfont\color{AcademicLevelTwo}}
  {\thesubsection}{0.70em}{}
\titlespacing*{\subsection}{0pt}{2.6ex plus .6ex minus .2ex}{0.95ex}

\titleformat{\subsubsection}
  {\normalfont\AcademicHeadingFont\bfseries\fontsize{11}{14}\selectfont\color{AcademicLevelThree}}
  {\thesubsubsection}{0.65em}{}
\titlespacing*{\subsubsection}{0pt}{2.2ex plus .5ex minus .2ex}{0.75ex}

% Shared running-head treatment: muted, sans-serif, no rule.
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}
\fancyhead[L]{\small\AcademicHeaderFont\color{AcademicMuted}\ReportShortTitle}
\fancyhead[R]{\small\color{AcademicMuted}\thepage}

\fancypagestyle{plain}{
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \fancyfoot[C]{\small\color{AcademicMuted}\thepage}
}
