% Layout and reusable typographic infrastructure.

\geometry{
  letterpaper,
  inner=31mm,
  outer=31mm,
  top=25mm,
  bottom=30mm,
  headheight=14pt,
  headsep=8mm,
  footskip=12mm
}

\setlength{\parindent}{1.2em}
\setlength{\parskip}{0pt}
\linespread{1.08}
\raggedbottom

\hypersetup{
  colorlinks=true,
  linkcolor=ThesisLink,
  citecolor=ThesisLink,
  urlcolor=ThesisLink,
  pdfauthor={\ThesisAuthor},
  pdftitle={\ThesisTitle}
}

\setcounter{tocdepth}{2}
\setcounter{secnumdepth}{3}

% Chapter architecture: large accent numeral, tracked uppercase heading, thin rule.
\titleformat{\chapter}[display]
  {\normalfont}
  {\makebox[\textwidth][r]{\color{ThesisAccent}\sffamily\fontsize{62}{62}\selectfont\thechapter}}
  {-8pt}
  {\color{ThesisText}\ThesisHeadingFont\fontsize{22}{28}\selectfont\MakeUppercase}
  [\vspace{0.7ex}{\color{ThesisRule}\titlerule}]

\titleformat{name=\chapter,numberless}[display]
  {\normalfont}
  {}
  {0pt}
  {\color{ThesisText}\ThesisHeadingFont\fontsize{22}{28}\selectfont\MakeUppercase}
  [\vspace{0.7ex}{\color{ThesisRule}\titlerule}]

\titlespacing*{\chapter}{0pt}{-6pt}{32pt}

\titleformat{\section}
  {\normalfont\color{ThesisAccent}\ThesisHeadingFont\large}
  {\thesection}
  {0.9em}
  {\MakeUppercase}
\titlespacing*{\section}{0pt}{3.2ex plus 1ex minus .2ex}{1.2ex}

\titleformat{\subsection}
  {\normalfont\color{ThesisAccent}\ThesisHeadingFont\normalsize}
  {\thesubsection}
  {0.8em}
  {}
\titlespacing*{\subsection}{0pt}{2.6ex plus .8ex minus .2ex}{0.8ex}

\titleformat{\subsubsection}
  {\normalfont\bfseries}
  {\thesubsubsection}
  {0.7em}
  {}

% Alternating running heads inspired by the reference dissertation.
\pagestyle{fancy}
\fancyhf{}
\renewcommand{\headrulewidth}{0.45pt}
\renewcommand{\footrulewidth}{0pt}
\fancyhead[LE]{\small\thepage}
\fancyhead[RE]{\small\ThesisSmallCapsFont\nouppercase{\rightmark}}
\fancyhead[LO]{\small\ThesisSmallCapsFont\nouppercase{\leftmark}}
\fancyhead[RO]{\small\thepage}

\renewcommand{\chaptermark}[1]{\markboth{#1}{}}
\renewcommand{\sectionmark}[1]{\markright{#1}}

\fancypagestyle{plain}{
  \fancyhf{}
  \renewcommand{\headrulewidth}{0pt}
  \fancyfoot[C]{\thepage}
}

% Captions.
\captionsetup{
  font=small,
  labelfont=bf,
  labelsep=colon,
  justification=centering,
  singlelinecheck=false
}

% Tables.
\renewcommand{\arraystretch}{1.12}

% Code style; monospaced font is controlled in typography.tex.
\lstdefinestyle{thesiscode}{
  basicstyle=\ttfamily\small,
  backgroundcolor=\color{ThesisCodeBackground},
  frame=single,
  rulecolor=\color{ThesisRule!35},
  numbers=left,
  numberstyle=\ttfamily\scriptsize\color{ThesisMuted},
  keywordstyle=\color{ThesisLink},
  commentstyle=\itshape\color{ThesisMuted},
  stringstyle=\color{ThesisAccent},
  breaklines=true,
  showstringspaces=false,
  tabsize=2
}
\lstset{style=thesiscode}

% Semantic helper commands.
\newcommand{\term}[1]{\emph{#1}}
\newcommand{\code}[1]{\texttt{#1}}
